\newglossaryentry{moa}{
    name={Maîtrise d'ouvrage (MOA)},
    description={
            C'est le client ou le demandeur du projet. Il définit les besoins, fixe ou non les objectifs, le budget et les délais, puis valide le résultat final
        }
}

\newglossaryentry{moe}{
    name={Maîtrise d'œuvre (MOE)},
    description={
            C'est l'équipe qui réalise le projet. Elle conçoit, développe et met en œuvre les solutions demandées par la MOA, en respectant les contraintes définies
        }
}

\newglossaryentry{metadonnees}{
    name={Métadonnées},
    text={métadonnées},
    description={
            Données qui décrivent d'autres données. Dans le contexte du projet, il s'agit des informations associées aux photos (date, heure, température, position GPS)
        }
}

\newglossaryentry{ihm}{
    name={Interface Homme-Machine (IHM)},
    description={
            Interface permettant à l'utilisateur d'interagir avec le système, notamment pour visualiser les données récupérées par le piège caméra
        }
}

\newglossaryentry{wp}{
    name={Work Package (WP)},
    description={
            Lot de travail, subdivision du projet permettant une répartition claire des tâches et une meilleure gestion des responsabilités
        }
}

\newglossaryentry{benchmark}{
    name={Benchmark},
    description={
            Étude comparative des différentes solutions techniques disponibles pour un composant ou une fonctionnalité du projet
        }
}
